+# -*- coding: utf-8 -*-
+"""真实 MCP 客户端：按标准协议连远端能力服务的 /mcp 入口。
+
+连接方式与 `hadoop/tools/mcp_client.py` 一致：带鉴权头的 HTTP 客户端 + 传输层 + 会话，
+每次动作开一次短会话，动作完成即关。任何连接/协议异常都翻成可读说明，不把堆栈甩给上层。
+
+红线：口令只放进请求头 `Authorization: Bearer <口令>`，**绝不**进日志或信封。
+不可用（无地址或无口令）时所有调用返回结构化 `MCP_UNAVAILABLE`，不发起任何网络请求。
+"""
+
+import asyncio
+import json
+
+from . import mcp_contract
+from .mcp_config import McpConfig
+
+#: 远端工具在工具表里的来源标记（契约层约定，全局唯一）。
+SOURCE_MCP = "mcp"
+
+
+def _flatten(exc):
+    """把异常组摊平：连接失败常被包成异常组，真正的原因藏在里层。"""
+    out = []
+    stack = [exc]
+    seen = set()
+    while stack:
+        item = stack.pop()
+        if id(item) in seen:
+            continue
+        seen.add(id(item))
+        out.append(item)
+        for sub in getattr(item, "exceptions", ()) or ():
+            stack.append(sub)
+    return out
+
+
+def explain(exc, url):
+    """把底层异常翻成能照着做的话（区分：口令不对 / 连不上 / 其它）。"""
+    flat = _flatten(exc)
+    text = "；".join("%s: %s" % (type(x).__name__, x) for x in flat[:3]) or str(exc)
+    low = " ".join(str(x).lower() for x in flat) or str(exc).lower()
+    if "401" in low or "unauthorized" in low:
+        return ("连接凭据不对：确认口令与请求头 Authorization: Bearer <口令>。"
+                "原始错误：%s" % text)
+    if "refused" in low or "connect" in low or "unreachable" in low:
+        return ("连不上 %s：服务可能没启动，或宿主机侧没配端口转发（虚拟机侧固定 8765）。"
+                "原始错误：%s" % (url, text))
+    return "调用失败：%s" % text
+
+
+class McpHttpClient(mcp_contract.McpClient):
+    """标准协议 MCP 客户端；`session_factory` 仅供测试注入（不连网）。"""
+
+    def __init__(self, config=None, session_factory=None):
+        self._config = config or McpConfig({})
+        self._session_factory = session_factory
+        self.interface_version = self._config.interface_version
+
+    @property
+    def config(self):
+        return self._config
+
+    # -- 内部 -----------------------------------------------------------------
+
+    def _envelope_error(self, code, message):
+        return {"ok": False, "interface_version": self.interface_version,
+                "error": {"code": code, "message": message}}
+
+    def _unavailable(self, action):
+        return self._envelope_error(
+            mcp_contract.CODE_MCP_UNAVAILABLE,
+            "远端能力服务未接入或未配置：%s" % action)
+
+    def _run(self, worker):
+        """开一次会话跑 `await worker(session)`；异常翻成 `{"_error": <可读说明>}`。
+
+        `worker` 是一个 `async def`，接收已握手的会话。
+        """
+        if self._session_factory is not None:
+            async def _fake():
+                session = self._session_factory()
+                await session.initialize()
+                return await worker(session)
+
+            try:
+                return asyncio.run(_fake())
+            except Exception as exc:                    # noqa: BLE001 —— 这里就是要报清楚
+                return {"_error": explain(exc, self._config.protocol_url())}
+
+        from mcp import ClientSession
+        from mcp.client.streamable_http import (
+            create_mcp_http_client, streamable_http_client)
+
+        headers = {"Authorization": "Bearer %s" % self._config.token}
+
+        async def _real():
+            async with create_mcp_http_client(headers=headers) as http_client:
+                async with streamable_http_client(
+                        self._config.protocol_url(), http_client=http_client) as streams:
+                    async with ClientSession(streams[0], streams[1]) as session:
+                        await session.initialize()
+                        return await worker(session)
+
+        try:
+            return asyncio.run(_real())
+        except Exception as exc:                        # noqa: BLE001 —— 这里就是要报清楚
+            return {"_error": explain(exc, self._config.protocol_url())}
+
+    @staticmethod
+    def _tool_row(tool):
+        schema = getattr(tool, "input_schema", None) or {}
+        return {"name": tool.name,
+                "title": getattr(tool, "title", None) or tool.name,
+                "description": getattr(tool, "description", "") or "",
+                "source": SOURCE_MCP,
+                "parameters": schema}
+
+    # -- 契约方法 -------------------------------------------------------------
+
+    def list_tools(self):
+        if not self._config.usable:
+            return []
+
+        async def worker(session):
+            result = await session.list_tools()
+            return [self._tool_row(tool)
+                    for tool in list(getattr(result, "tools", []) or [])]
+
+        rows = self._run(worker)
+        return rows if isinstance(rows, list) else []
+
+    def call_tool(self, name, args):
+        if not self._config.usable:
+            return self._unavailable("call_tool(%s)" % name)
+
+        async def worker(session):
+            result = await session.call_tool(name, args or {})
+            texts = []
+            for item in getattr(result, "content", []) or []:
+                text = getattr(item, "text", None)
+                if text is not None:
+                    texts.append(text)
+            return "\n".join(texts)
+
+        text = self._run(worker)
+        if isinstance(text, dict) and text.get("_error"):
+            return self._envelope_error(mcp_contract.CODE_DRIVER_UNREACHABLE,
+                                        text["_error"])
+        try:
+            payload = json.loads(text)
+        except (ValueError, TypeError):
+            return self._envelope_error(
+                mcp_contract.CODE_DRIVER_BAD_OUTPUT,
+                "远端返回非信封：%s" % str(text)[:300])
+        if not isinstance(payload, dict):
+            return self._envelope_error(mcp_contract.CODE_DRIVER_BAD_OUTPUT,
+                                        "远端返回非对象信封")
+        return payload
+
+    def get_task_status(self, task_id):
+        return self.call_tool("get_task_status", {"task_id": task_id})
+
+    def describe_capabilities(self):
+        return self.call_tool("describe_capabilities", {})
+
+    def health(self):
+        """健康检查：不进协议入口，直接 GET /health（免口令）。"""
+        if not self._config.usable:
+            return self._unavailable("health")
+        import httpx
+        try:
+            resp = httpx.get(self._config.health_url(),
+                             timeout=self._config.timeout_sec)
+            data = resp.json()
+            return data if isinstance(data, dict) else {"ok": True, "raw": data}
+        except Exception as exc:                        # noqa: BLE001 —— 这里就是要报清楚
+            return self._envelope_error(
+                mcp_contract.CODE_DRIVER_UNREACHABLE,
+                explain(exc, self._config.health_url()))

+# -*- coding: utf-8 -*-
+"""真实 MCP 客户端（agent/mcp_http_client.py）单元测试。
+
+全部走注入的 `session_factory` 接缝：不连网、不起服务、不触模型。
+覆盖：列工具映射、调工具信封解析、非信封兜底、配置不可用、连接异常翻译、
+口令绝不外泄、以及基类接口（版本协商 / 关闭）。
+"""
+import json
+import os
+import sys
+
+AGENT_DIR = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
+REPO_ROOT = os.path.dirname(AGENT_DIR)
+sys.path.insert(0, REPO_ROOT)
+
+from agent import mcp_contract  # noqa: E402
+from agent import mcp_http_client as mc  # noqa: E402
+
+
+def cfg(url="http://127.0.0.1:8900/mcp", token="t" * 20):
+    return mc.McpConfig({"url": url, "token": token})
+
+
+class FakeSession:
+    """假会话：不连网，按预置结果返回，并记录调用。"""
+
+    def __init__(self, tools=None, call_result=None, call_text=None):
+        self._tools = tools or []
+        self._call_result = call_result
+        self._call_text = call_text
+        self.calls = []
+
+    async def initialize(self):
+        return None
+
+    async def list_tools(self):
+        class _T:
+            def __init__(self, name, schema):
+                self.name = name
+                self.title = name
+                self.input_schema = schema
+
+        return type("R", (), {"tools": [_T(n, s) for n, s in self._tools]})()
+
+    async def call_tool(self, name, args):
+        class _C:
+            def __init__(self, text):
+                self.text = text
+
+        self.calls.append((name, dict(args or {})))
+        text = (self._call_text if self._call_text is not None
+                else json.dumps(self._call_result))
+        return type("R", (), {"content": [_C(text)]})()
+
+
+# ---------------------------------------------------------------------------
+# 列工具
+# ---------------------------------------------------------------------------
+
+def test_list_tools_maps_to_rows():
+    client = mc.McpHttpClient(cfg(), session_factory=lambda: FakeSession(
+        tools=[("list_schemes", {"type": "object", "properties": {}, "required": []})]))
+    rows = client.list_tools()
+    assert rows[0]["name"] == "list_schemes"
+    assert rows[0]["source"] == "mcp"
+    assert rows[0]["parameters"]["type"] == "object"
+
+
+def test_list_tools_unusable_returns_empty():
+    client = mc.McpHttpClient(mc.McpConfig({}))
+    assert client.list_tools() == []
+
+
+# ---------------------------------------------------------------------------

+def test_call_tool_returns_envelope():
+    client = mc.McpHttpClient(cfg(), session_factory=lambda: FakeSession(
+        call_result={"ok": True, "interface_version": "1.0", "task_id": "T1"}))
+    env = client.call_tool("get_task_status", {"task_id": "T1"})
+    assert env["ok"] is True and env["task_id"] == "T1"
+
+
+def test_call_tool_passes_name_and_args():
+    session = FakeSession(call_result={"ok": True})
+    client = mc.McpHttpClient(cfg(), session_factory=lambda: session)
+    client.call_tool("get_samples", {"task_id": "H1", "limit": 3})

+def test_call_tool_non_json_is_bad_output():

+    assert env["error"]["code"] == mcp_contract.CODE_DRIVER_BAD_OUTPUT

+
+def test_connection_refused_is_explained_and_token_hidden():
+    def boom():
+        raise ConnectionRefusedError("connection refused")
+

+    assert env["error"]["code"] == mcp_contract.CODE_DRIVER_UNREACHABLE
+    assert "连不上" in env["error"]["message"]
+    assert "super-secret-token" not in env["error"]["message"]
+
+
+def test_http_401_is_explained_as_credential_problem():
+    def boom():
+        raise RuntimeError("unexpected status: 401 Unauthorized")
+

+    env = client.call_tool("x", {})
+    assert env["error"]["code"] == mcp_contract.CODE_DRIVER_UNREACHABLE
+    assert "凭据" in env["error"]["message"]
+    assert "super-secret-token" not in env["error"]["message"]
+
+
+def test_generic_error_is_explained():
+    def boom():
+        raise RuntimeError("some other protocol failure")
+
+    client = mc.McpHttpClient(cfg(), session_factory=boom)
+    env = client.call_tool("x", {})
+    assert env["error"]["code"] == mcp_contract.CODE_DRIVER_UNREACHABLE
