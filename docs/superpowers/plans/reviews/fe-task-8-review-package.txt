 /** 把错误码翻成中文；未知码给可读兜底（保留原码便于排查）。 */
 export function errorCodeText(code: string | null | undefined): string {
   if (!code) return '未知错误'
   const hit = ERROR_CODE_TEXT[code]
   if (hit) return hit
   if (code.startsWith('HTTP_')) return `请求失败（HTTP ${code.slice(5)}）`
   return `未知错误（${code}）`
 }
