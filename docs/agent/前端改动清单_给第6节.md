## 零、先看这段：后端改了什么，对你有什么影响

### 1. 「追问丢口径」这个 bug，**后端已经修好了，你可以一行不改**

后端新增了**按 task_id 记录运行口径的快照**（`agent/session.py`）。
追问本来就带 `task_id`，所以后端能自己找回第一轮选定的 `exec_mode` / `scope`。

**你现在不改也不会坏。** 下面 F1 只是"加固"（让参数显式回传），有余力再做。

### 2. `/api/chat` 返回体多了 4 个字段（加在外层，`data` 结构没变）

```json
{
  "ok": true, "intent": "...", "intent_cn": "...", "reply": "...",
  "data": {...}, "task_id": "...",
  "task_started": true,
  "opts": {"exec_mode": "cluster", "scope": "full"},
  "engine": "rules",
  "llm": {"configured": false, "used": false, "note": ""}
}
```

| 字段 | 建议怎么用 |
|---|---|
| `task_started` | 为 `true` 时才开始轮询进度（以前靠 intent 猜） |
| `opts` | 本轮**真实生效**的口径，可直接回显；下次想显式回传就用它 |
| `engine` | `"rules"` / `"llm"`。**建议显示在助手气泡上**（`Agent · 规则解析`），老师问"你们到底接没接大模型"时一眼能答 |
| `llm.note` | 回落原因（如"网络不可达…已回落到规则解析"），原样显示即可 |

### 3. 端点 10 → 11

新增 `POST /api/llm/test`（大模型连通性自检）。`/health` 多了 `llm.supported` / `llm.configured`。

> **关键**：`llm.supported=false` 时，**页面上不应出现任何大模型相关 UI**（齿轮按钮保持 hidden）。

### 4. 一个需要跟队长对齐的数字

样本档是 **评分表前 2000 行**（`hadoop/driver/run_task.py:744` `--sample 2000`），
**不是 400 行**。如果听到"四百来份样本"的说法，以 2000 为准。

### F3-③ `frontend/js/app.js`：`boot()` 之前新增 `initResizer()`

```js
  /* ------------------------------------------------------------- 分栏拖拽 */

  var store = {
    get: function (k) {
      try { return window.localStorage.getItem(k); } catch (e) { return null; }
    },
    set: function (k, v) {
      try { window.localStorage.setItem(k, String(v)); } catch (e) { /* file:// 下忽略 */ }
    }
  };

  function initResizer() {
    var bar = $('rail-resizer');
    if (!bar) return;

    var RAIL_MIN = 320, RAIL_MAX = 720, RAIL_KEY = 'od.rail-w';
    var dragging = false;

    function twoColumn() { return window.matchMedia('(min-width: 1101px)').matches; }

    function clampWidth(px) {
      // 右栏至少留 520px 的内容宽度
      var max = Math.max(RAIL_MIN, Math.min(RAIL_MAX, window.innerWidth - 520));
      return Math.max(RAIL_MIN, Math.min(max, px));
    }

    function setWidth(px, persist) {
      document.documentElement.style.setProperty('--rail-w', px + 'px');
      if (persist) store.set(RAIL_KEY, px);
    }

    function currentWidth() {
      var v = parseInt(
        getComputedStyle(document.documentElement).getPropertyValue('--rail-w'), 10);
      return isNaN(v) ? 400 : v;
    }

    function clientX(e) {
      return (e.touches && e.touches.length) ? e.touches[0].clientX : e.clientX;
    }

    function onMove(e) {
      if (!dragging) return;
      var rect = document.querySelector('.shell').getBoundingClientRect();
      setWidth(clampWidth(clientX(e) - rect.left), false);
      e.preventDefault();
    }

    function stopDrag() {
      if (!dragging) return;
      dragging = false;
      bar.classList.remove('is-active');
      document.removeEventListener('mousemove', onMove);
      document.removeEventListener('mouseup', stopDrag);
      store.set(RAIL_KEY, currentWidth());
    }

    function startDrag(e) {
      if (!twoColumn()) return;          // 1100px 以下已塌成单栏，禁用拖拽
      dragging = true;
      bar.classList.add('is-active');
      document.addEventListener('mousemove', onMove);
      document.addEventListener('mouseup', stopDrag);
      e.preventDefault();
    }

    bar.addEventListener('mousedown', startDrag);
    bar.addEventListener('touchstart', startDrag);
    document.addEventListener('touchmove', onMove);
    document.addEventListener('touchend', stopDrag);

    // 键盘可达：←/→ 每次 16px
    bar.addEventListener('keydown', function (e) {
      if (!twoColumn()) return;
      var cur = currentWidth();
      if (e.key === 'ArrowLeft') { e.preventDefault(); setWidth(clampWidth(cur - 16), true); }
      else if (e.key === 'ArrowRight') { e.preventDefault(); setWidth(clampWidth(cur + 16), true); }
    });

    window.addEventListener('resize', function () {
      if (twoColumn()) setWidth(clampWidth(currentWidth()), false);
    });

    var saved = parseInt(store.get(RAIL_KEY), 10);
    if (saved) setWidth(clampWidth(saved), false);
  }
```

### F4-⑤ 新文件 `frontend/js/llm-settings.js`

```js
/* =============================================================================
   llm-settings.js —— 大模型「可选增强」设置面板
   -----------------------------------------------------------------------------
   三条纪律：
     1. 后端不支持（/health 里 llm.supported 不为 true）⇒ 页面上完全不出现
        任何大模型相关元素，齿轮按钮保持 hidden。
     2. 配置只写 localStorage，**不写任何仓库文件**，也不会被后端持久化。
     3. 每次会话请求才随 body.llm 传给后端；清空后等价于「从未接过大模型」。
   ========================================================================== */

(function (global) {
  'use strict';

  var KEY = 'od.llm';
  var $ = function (id) { return document.getElementById(id); };

  var cfg = { enabled: false, api_base: '', api_key: '', model: '' };

  function load() {
    try {
      var raw = global.localStorage.getItem(KEY);
      if (!raw) return null;
      var o = JSON.parse(raw);
      if (!o || typeof o !== 'object') return null;
      cfg.enabled = !!o.enabled;
      cfg.api_base = o.api_base || '';
      cfg.api_key = o.api_key || '';
      cfg.model = o.model || '';
      return cfg;
    } catch (e) { return null; }
  }

  function save() {
    try { global.localStorage.setItem(KEY, JSON.stringify(cfg)); return true; }
    catch (e) { setStatus('本机无法保存（浏览器限制），本次会话内仍生效。'); return false; }
  }

  function setStatus(msg) { $('llm-status').textContent = msg || ''; }

  function fillForm() {
    $('llm-enabled').checked = cfg.enabled;
    $('llm-base').value = cfg.api_base;
    $('llm-key').value = cfg.api_key;
    $('llm-model').value = cfg.model;
  }

  function readForm() {
    cfg.enabled = $('llm-enabled').checked;
    cfg.api_base = ($('llm-base').value || '').trim();
    cfg.api_key = ($('llm-key').value || '').trim();
    cfg.model = ($('llm-model').value || '').trim();
    return cfg;
  }

  /** 供 app.js 在每次 chat 时取用；未启用或信息不全一律返回 null。 */
  function payload() {
    if (!cfg || !cfg.enabled) return null;
    if (!cfg.api_base || !cfg.api_key || !cfg.model) return null;
    return { enabled: true, api_base: cfg.api_base, api_key: cfg.api_key, model: cfg.model };
  }

  /** 后端不支持就整块隐藏 —— 「没配置时完全不显示大模型相关 UI」。 */
  function syncVisibility(health) {
    var llm = (health && health.llm) || null;
    var supported = !!(llm && llm.supported);
    $('btn-settings').hidden = !supported;
    if (!supported) $('settings-panel').hidden = true;
    return supported;
  }

  function openPanel() {
    load(); fillForm();
    $('settings-panel').hidden = false;
    setStatus('');
  }

  function bind() {
    $('btn-settings').addEventListener('click', function () {
      if ($('settings-panel').hidden) openPanel(); else $('settings-panel').hidden = true;
    });
    $('btn-settings-close').addEventListener('click', function () { $('settings-panel').hidden = true; });

    $('btn-llm-save').addEventListener('click', function () {
      readForm();
      if (cfg.enabled && (!cfg.api_base || !cfg.api_key || !cfg.model)) {
        setStatus('三个字段都要填；不全的话本次不会启用，走规则解析。');
        return;
      }
      save();
```

## F5 ｜「AI 风味」措辞替换建议

> **先说结论**：我全目录搜过"让我想想/正在分析/已为你/思考中"这类小短语，**一个都没有**。
> 你看到的"AI 味"其实是**结构性的人格化**：整个左栏围绕"Agent"这个人称组织。共 10 处。

### ⚠️ 一条重要警告

**不要写「让我想想 / AI 正在分析 / 正在思考中」这类文案。**
规则路径下**没有思考过程**，写了就是在冒充智能，老师一问就穿。

**正确的拟人化是"它在动"而不是"它在想"**——目前真正缺的是一次对话中的**中间态**：
点了发送到 Agent 回复之间是静默的。补上"已收到，正在调用工具…"比改称呼有用得多。

### #8 的代码（`app.js:1132` pushChat 改造）

**现在（`:1137-1139`）：**

```js
    var msg = el('div', 'msg msg--' + (who === 'user' ? 'user' : 'agent'));
```

**改成：**

## F6 ｜「全量 / 样本」选择控件 —— **建议不加**

| 判断项 | 结论 |
|---|---|
| 现在链路完整吗？ | ✅ 完整。driver 默认 `full`；真有人跑了 sample，`app.js:421-456 checkSampling()` 也会按真实行数挂黄横幅「抽样运行，数字不可用于汇报」 |
| 加了的风险 | ⚠️ 在 UI 上放一个"一键产出不能汇报的数字"的按钮，跟课程引言 §13/§90 的红线正面对撞；**汇报现场误点率不为零** |
| 收益 | 只有"联调快一点"。而本地引擎（约 80 秒）已经够快，不必再用 sample |
| 前置依赖 | 需要先确认口径档位（**是 2000 行，不是 400 行**） |

**如果组长坚持要加**，建议做成**隐藏的开发选项**（默认不展示，URL 带 `?debug=1` 才显示）：

```html

        <label class="radio"><input type="radio" name="run-scope" value="full" checked /><span><strong>全量</strong><em>正式口径，约 10 分钟</em></span></label>
        <label class="radio"><input type="radio" name="run-scope" value="sample" /><span><strong>样本</strong><em>评分表前 2000 行，数字不可用于汇报</em></span></label>
      </fieldset>
```

```js
    if (/[?&]debug=1/.test(location.search)) $('scope-row').hidden = false;
```

并配一个 `currentScope()`，把 F1 里 `API.chat(..., null, ...)` 的第四个参数换成它。
